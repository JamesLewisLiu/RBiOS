cmake_minimum_required(VERSION 3.20)
project(RBiOS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB RBCORE_SOURCES CONFIGURE_DEPENDS core/src/*.cpp)
add_library(rbcore ${RBCORE_SOURCES})
target_include_directories(rbcore PUBLIC core/include)
if(MSVC)
  target_compile_options(rbcore PRIVATE /W4 /fp:strict)
else()
  target_compile_options(rbcore PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
add_executable(rbtests tests/CoreTests.cpp)
target_link_libraries(rbtests PRIVATE rbcore)
add_executable(rbinspect tools/Inspect.cpp)
target_link_libraries(rbinspect PRIVATE rbcore)
enable_testing()
add_test(NAME core COMMAND rbtests)
foreach(component IN ITEMS Motion Protocol Score JudgeDispatch Touch CPU Gameplay AFP MusicSheet NoteFrame Route NoteGraphic TouchInput StageScoreInitialization StageSession AfpTimeline StageGraphic LongNoteGraphic OriginalAfpMoviePool GamePresentation StageLongPresentation StageScoreGraphic ScoreDisplay StageLabelGraphic StageSound)
  string(TOLOWER "${component}" lower)
  add_executable(rb${lower} tests/${component}Tests.cpp)
  target_link_libraries(rb${lower} PRIVATE rbcore)
  add_test(NAME ${lower} COMMAND rb${lower})
endforeach()
